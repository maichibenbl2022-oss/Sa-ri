local KEY_TAIL     = ".  "
local KEY_DURATION = 3600
local KEY_FILE     = "HM_key70.dat"

local function readKeyRaw()
    local ok, d = pcall(function()
        if readfile and isfile and isfile(KEY_FILE) then 
            return readfile(KEY_FILE) 
        end
        return nil
    end)
    return ok and d or nil
end

local function writeKeyRaw(s) 
    pcall(function() 
        if writefile then 
            writefile(KEY_FILE, s) 
        end 
    end) 
end

local function wipeKey() 
    pcall(function() 
        if writefile then 
            writefile(KEY_FILE, "") 
        end 
    end) 
end

local function validateKey(k)
    if type(k) ~= "string" or #k < 4 then 
        return false 
    end
    if string.sub(k, -3) ~= KEY_TAIL then 
        return false 
    end
    return true
end

local function keyValidNow()
    local raw = readKeyRaw()
    if not raw or raw == "" then 
        return false 
    end
    local a, b = string.find(raw, "|", 1, true)
    if not a then 
        return false 
    end
    local k  = string.sub(raw, 1, a - 1)
    local ts = tonumber(string.sub(raw, b + 1))
    if not k or not ts or not validateKey(k) then 
        return false 
    end
    if (os.time() - ts) >= KEY_DURATION then 
        return false 
    end
    return true
end

local function Valid(p)
    if not p or p == LocalPlayer or Allies[p.Name] then 
        return false 
    end
    local c = p.Character
    if not c then 
        return false 
    end
    local r = c:FindFirstChild("HumanoidRootPart")
    local h = c:FindFirstChildOfClass("Humanoid")
    return r and h and h.Health > 0
end

local function FindNearest(range)
    local mr = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
    if not mr then 
        return nil 
    end
    local best, bd = nil, range or 100
    for _, p in ipairs(Players:GetPlayers()) do
        if Valid(p) then
            local r = p.Character:FindFirstChild("HumanoidRootPart")
            if r then
                local d = (mr.Position - r.Position).Magnitude
                if d < bd then 
                    bd = d
                    best = p 
                end
            end
        end
    end
    return best
end

local function FindByName(name)
    if not name or name == "" then 
        return nil 
    end
    name = string.lower(name)
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= LocalPlayer and string.find(string.lower(p.Name), name, 1, true) then 
            return p 
        end
    end
    return nil
end

local function GetTool()
    local c = LocalPlayer.Character
    if not c then 
        return nil 
    end
    local t = c:FindFirstChildOfClass("Tool")
    if t then 
        return t 
    end
    local bp = LocalPlayer:FindFirstChild("Backpack")
    if bp then
        for _, x in ipairs(bp:GetChildren()) do
            if x:IsA("Tool") then 
                x.Parent = c
                return x 
            end
        end
    end
    return nil
end

local function HitAll(target)
    local c = LocalPlayer.Character
    if not c then 
        return 
    end
    for _, t in ipairs(c:GetChildren()) do
        if t:IsA("Tool") then 
            pcall(function() 
                t.Enabled = true
                t:Activate() 
            end) 
        end
    end
    for _, r in ipairs(Remotes) do
        pcall(function() r:FireServer("M1") end)
        pcall(function() r:FireServer() end)
    end
    for _, r in ipairs(ManualRemotes) do
        pcall(function() r:FireServer("M1") end)
        pcall(function() r:FireServer("Attack") end)
        pcall(function() r:FireServer() end)
    end
    if clientSideHitbox and target then 
        pcall(function() 
            clientSideHitbox:FireServer(target, buffer.fromstring("10069.74")) 
        end) 
    end
    if combatRemote then 
        pcall(function() combatRemote:FireServer("M1") end) 
        pcall(function() combatRemote:FireServer("Attack") end) 
    end
end

-- 2. OR LOOP
task.spawn(function()
    while true do
        task.wait(0.02)
        if State.ORLock and not State.Farm then
            pcall(function()
                if not Valid(Config.Target) then
                    local name = Config.TargetName
                    if not name or name == "" then 
                        name = TargetBox.Text
                        Config.TargetName = name 
                    end
                    Config.Target = (name == "") and FindNearest(Config.ORRange) or FindByName(name)
                    if not Config.Target then 
                        ORLastTarget = nil
                        return 
                    end
                end

                local tc = Config.Target.Character
                local tRoot = tc and tc:FindFirstChild("HumanoidRootPart")
                local myC = LocalPlayer.Character
                local myRoot = myC and myC:FindFirstChild("HumanoidRootPart")
                local myHum = myC and myC:FindFirstChildOfClass("Humanoid")
                if not (tRoot and myRoot and myHum) then 
                    return 
                end
                local dist = (myRoot.Position - tRoot.Position).Magnitude
                if dist > 2000 then 
                    return 
                end

                local tp = tRoot.Position
                local now = tick()

                if Config.ORMode == "DASH" then
                    local function tpToTarget()
                        local behind = tp - tRoot.CFrame.LookVector * Config.DashOrbitRadius
                        myRoot.CFrame = CFrame.lookAt(Vector3.new(behind.X, tp.Y, behind.Z), tp)
                    end
                    if dashPhase == "idle" then
                        dashSavedPos = myRoot.CFrame
                        dashOrbitAngle = math.random() * math.pi * 2
                        tpToTarget()
                        dashPhase = "attack"
                        dashTimer = now + Config.DashAttackDuration
                        dashNextHit = 0
                    elseif dashPhase == "attack" then
                        dashOrbitAngle = dashOrbitAngle + math.rad(Config.DashOrbitSpeed) * 0.02
                        local px = tp.X + math.cos(dashOrbitAngle) * Config.DashOrbitRadius
                        local pz = tp.Z + math.sin(dashOrbitAngle) * Config.DashOrbitRadius
                        myRoot.CFrame = CFrame.lookAt(Vector3.new(px, tp.Y + Config.DashGroundOffset, pz), tp)
                        
                        if now >= dashNextHit then 
                            dashNextHit = now + Config.DashHitRate
                            HitAll(Config.Target) 
                        end
                        if now >= dashTimer then
                            if dashSavedPos then 
                                myRoot.CFrame = dashSavedPos 
                            end
                            dashPhase = "wait"
                            dashTimer = now + Config.DashWaitDuration
                        end
                    elseif dashPhase == "wait" then
                        if now >= dashTimer then 
                            tpToTarget()
                            dashPhase = "attack"
                            dashTimer = now + Config.DashAttackDuration
                            dashNextHit = 0 
                        end
                    end
                elseif Config.ORMode == "TPLOOP" then
                    if dashPhase == "idle" then
                        dashSavedPos = myRoot.CFrame
                        local behind = tp - tRoot.CFrame.LookVector * Config.ORDistance
                        myRoot.CFrame = CFrame.lookAt(Vector3.new(behind.X, tp.Y, behind.Z), tp)
                        dashPhase = "attack"
                        dashTimer = now + Config.TPLoopTime
                    elseif dashPhase == "attack" then
                        HitAll(Config.Target)
                        if now >= dashTimer then
                            myRoot.CFrame = dashSavedPos
                            dashPhase = "wait"
                            dashTimer = now + 1
                        end
                    elseif dashPhase == "wait" then
                        if now >= dashTimer then 
                            dashPhase = "idle" 
                        end
                    end
                elseif Config.ORMode == "ORBIT360" then
                    Config.orbitAngle = Config.orbitAngle + math.rad(Config.ORRotSpeed) * 0.05
                    local px = tp.X + math.cos(Config.orbitAngle) * Config.ORDistance
                    local pz = tp.Z + math.sin(Config.orbitAngle) * Config.ORDistance
                    local rp = RaycastParams.new()
                    rp.FilterDescendantsInstances = {myC, tc}
                    local hit = workspace:Raycast(Vector3.new(px, tp.Y + 10, pz), Vector3.new(0, -30, 0), rp)
                    local gy = hit and hit.Position.Y or (tp.Y - 3)
                    myRoot.CFrame = CFrame.lookAt(Vector3.new(px, gy + 3, pz), tp)
                    if dist <= Config.ORNear + 2 then 
                        HitAll(Config.Target) 
                    end
                else
                    local fp = tp - tRoot.CFrame.LookVector * Config.ORDistance
                    myRoot.CFrame = CFrame.lookAt(fp + Vector3.new(0, Config.ORHeight, 0), tp)
                    if dist <= Config.ORNear then 
                        HitAll(Config.Target) 
                    end
                end
            end)
        end
    end
end)

                                local cHum = p.Character:FindFirstChild("Humanoid")

                                    local dx, dy, dz = cRoot.Position.X - myPos.X, cRoot.Position.Y - myPos.Y, cRoot.Position.Z - myPos.Z
                                    if (dx*dx+dy*dy+dz*dz) <= COMBAT_RANGE_SQ then
                                        local anims = cHum:GetPlayingAnimationTracks()
                                        for i = 1, #anims do 
                                            if M1_Hash[anims[i].Animation.AnimationId] then 
                                                m1Detected = true
                                                break 
                                            end 
                                        end
                                    end
                                end
                                if m1Detected then 
                                    break 
                                end
                            end
                        end
                        if m1Detected then 
                            lastM1Time = tick()
                            rawBlock() 
                        elseif isBlocking and (tick() - lastM1Time) > BUFFER_TIME then

                        end

            local c = LocalPlayer.Character
            if c then 
                local hrp = c:FindFirstChild("HumanoidRootPart")
                if hrp then 
                    hrp.CFrame = hrp.CFrame * CFrame.Angles(0, math.rad(15), 0) 
                end 
            end 
        end
        
        if State.AntiStun then 
            local h = LocalPlayer.Character and LocalPlayer.Character:FindFirstChildOfClass("Humanoid")
            if h then 
                h.PlatformStand = false
                h.Sit = false
                h.AutoRotate = true 
            end 
        end
        
        if State.Dash then 
            local h = LocalPlayer.Character and LocalPlayer.Character:FindFirstChildOfClass("Humanoid")
            if h then 
                h.WalkSpeed = Config.AntiDetect and 60 or 200
                h.JumpPower = 80 
            end 
        end
        
        if State.GodMode then 
            local c = LocalPlayer.Character
            if c then 
                local h = c:FindFirstChildOfClass("Humanoid")
                if h then 
                    h.MaxHealth = math.huge
                    h.Health = math.huge 
                end 
            end 
        end
        
        if State.NoClip then 
            local c = LocalPlayer.Character
            if c then 
                for _, part in ipairs(c:GetDescendants()) do 
                    if part:IsA("BasePart") then 
                        part.CanCollide = false 
                    end 
                end
                local hrp = c:FindFirstChild("HumanoidRootPart")
                if hrp and hrp.Position.Y < -50 then 
                    hrp.CFrame = CFrame.new(hrp.Position.X, 50, hrp.Position.Z) 
                end 
            end 
        end
        
        if State.Aim and Config.ORMode ~= "DASH" and Config.ORMode ~= "EZY" then 
            local t = State.ORLock and Valid(Config.Target) and Config.Target or FindNearest(Config.AttackRange)
            if t then 
                local mr = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
                local tr = t.Character and t.Character:FindFirstChild("HumanoidRootPart")
                if mr and tr then 
                    mr.CFrame = CFrame.lookAt(mr.Position, tr.Position) 
                end 
            end 
        end
        
        if State.TP and not State.ORLock then 
            local t = FindNearest(Config.AttackRange)
            if t then 
                local mr = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
                local tr = t.Character and t.Character:FindFirstChild("HumanoidRootPart")
                if mr and tr then 
                    mr.CFrame = tr.CFrame * CFrame.new(0, 0, Config.TPDistance) 
                end 
            end 
        end

        for _, p in ipairs(Players:GetPlayers()) do
            if Valid(p) then
                local pc = p.Character
                if State.ESP then
                    if not pc:FindFirstChild("ESP_HL") then 
                        local hl = Instance.new("Highlight")
                        hl.Name = "ESP_HL"
                        hl.FillColor = Color3.fromRGB(0, 255, 100)
                        hl.OutlineColor = Color3.fromRGB(255, 255, 255)
                        hl.FillTransparency = 0.5
                        hl.Parent = pc 
                    end
                else
                    local hl = pc:FindFirstChild("ESP_HL")
                    if hl then 
                        hl:Destroy() 
                    end
                end
            end
        end
    end)
end)

-- 5. RENDER (HITBOX & FLY)
RunService.RenderStepped:Connect(function()
    pcall(function()
        if State.Hitbox then
            for _, p in ipairs(Players:GetPlayers()) do
                if p ~= LocalPlayer and p.Character then
                    local r = p.Character:FindFirstChild("HumanoidRootPart")
                    if r then 
                        r.Size = Vector3.new(Config.HitboxSize, Config.HitboxSize, Config.HitboxSize)
                        r.Transparency = 0.7
                        r.CanCollide = false 
                    end
                end
            end
        end
        if State.Fly and not State.Farm then
            local c = LocalPlayer.Character
            local r = c and c:FindFirstChild("HumanoidRootPart")
            local h = c and c:FindFirstChild("Humanoid")
            if r and h then 
                r.AssemblyLinearVelocity = h.MoveDirection * Config.FlySpeed + Vector3.new(0, 2, 0) 
            end
        end
    end)
end)

-- 6. AURA
local AuraFolder = nil
task.spawn(function()
    while true do
        task.wait(0.1)
        pcall(function()
            if State.Aura then
                local r = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
                if r then
                    if not AuraFolder or not AuraFolder.Parent then
                        if AuraFolder then 
                            AuraFolder:Destroy() 
                        end
                        AuraFolder = Instance.new("Folder")
                        AuraFolder.Name = "PlayerAura"
                        AuraFolder.Parent = workspace
                    end
                    local body = AuraFolder:FindFirstChild("BodyAura")
                    if not body then
                        body = Instance.new("Part")
                        body.Name = "BodyAura"
                        body.Shape = Enum.PartType.Ball
                        body.Material = Enum.Material.Neon
                        body.CanCollide = false
                        body.Massless = true
                        body.Parent = AuraFolder
                        local wc = Instance.new("WeldConstraint")
                        wc.Part0 = body
                        wc.Part1 = r
                        wc.Parent = body
                    end
                    body.Size = Vector3.new(8, 8, 8)
                    body.Color = Color3.fromRGB(255, 0, 0)
                    body.Transparency = 0.5
                    body.CFrame = r.CFrame
                end
            else
                if AuraFolder then 
                    AuraFolder:Destroy()
                    AuraFolder = nil 
                end
            end
        end)
    end
end)
